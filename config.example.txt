# Real Drift (realdrift_nx) config

# Driving controls (only while a car is driven; menus use touch / the cursor).
# One or more of: A B X Y L R ZL ZR PLUS MINUS UP DOWN LEFT RIGHT LSTICK RSTICK,
# separated by commas; NONE to unbind. The left stick always steers.
bind_throttle=ZR
bind_brake=ZL
bind_handbrake=B
bind_clutch=Y
bind_gear_up=UP
bind_gear_down=DOWN
bind_camera=NONE
bind_steer_left=LEFT
bind_steer_right=RIGHT
# left stick deadzone, 0.0 .. 0.9
stick_deadzone=0.15
# pause / resume the race (also closes the pause menu)
bind_pause=PLUS
# 1 hides the on-screen steering/throttle/brake/handbrake/gear buttons in races
hide_touch_controls=1

# Menus.
# menu_navigation: 1 = a frame marks a menu element; the D-pad or the left stick moves it,
#   bind_ui_select taps it, bind_ui_back goes back. The touchscreen keeps working.
#   0 = the on-screen cursor of earlier versions (stick moves it, A taps, + hides it).
menu_navigation=1
bind_ui_select=A
bind_ui_back=B
# bind_ui_prev / bind_ui_next: previous / next car in the garage, tab, leaderboard track.
#   bind_ui_select and bind_ui_back also skip the track preview before a race.
bind_ui_prev=L
bind_ui_next=R

# Performance.
# render_resolution: auto = 720 handheld / 1080 docked (picked at launch), or 720 / 1080.
#   Handheld shows 720p anyway; rendering 1080p there costs 2.25x pixels (smoke!).
render_resolution=auto
# boost_loading: 1 raises the CPU clock (1785 MHz) while a scene is loading.
boost_loading=1
# Heavy tyre smoke is what drops the frame rate (the GPU, not the CPU). Both keys at
# 100 = the original game.
# smoke_size: size of the smoke puffs in percent, 10 .. 100 (try 50).
smoke_size=100
# smoke_amount: percent of the smoke puffs that are emitted, 10 .. 100 (try 50).
smoke_amount=100
# texture_transcode: 1 stores the game's ETC2 textures as S3TC, which the GPU reads
#   directly; 0 leaves them to mesa, which unpacks each one to 4-8x the memory, so after
#   many cars and tracks new textures stop fitting and show black. Loads take a bit longer.
texture_transcode=1
# archive_cache: 1 keeps recently read parts of base.apk and the OBB in memory (64 MB),
#   shared by every open of them, so loads read each part from the SD card once; 0 reads
#   the card directly (slower loads).
archive_cache=1
# debug_log: 1 = full debug.log (send it when reporting problems), 0 = only crashes,
#   stalls, GC and startup lines. The log is buffered either way.
debug_log=0
# debug_fps_counter: 1 shows frames per second and the slowest frame (ms) of the last
#   half second in the top-left corner.
debug_fps_counter=0
# debug_gl_trace: 1 logs where a frame's GPU time goes, per render target ([gltrace]
#   lines, with debug_log = 1), about every 10 s. Frames run slower while one is taken.
debug_gl_trace=0
# debug_ui_nav: 1 logs the elements menu navigation finds on each new screen ([uinav]
#   lines, with debug_log = 1): names, parents, rectangles.
debug_ui_nav=0
# debug_mem: 1 logs where the memory is about every 10 s and after each load, and every
#   GL error when a texture or buffer is created ([mem] lines, with debug_log = 1).
debug_mem=0
# debug_load: 1 logs, for every frame of a second or more (a load), the file traffic the
#   engine asked for and what reached the SD card, and each engine thread's CPU time
#   ([load] lines, with debug_log = 1).
debug_load=0
# debug_audio: 1 logs when the sound goes silent for 3 s or comes back, when the audio
#   callback stops, and the engine's volume / mute calls ([audio] lines, with debug_log = 1).
debug_audio=0
